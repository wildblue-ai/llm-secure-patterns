def budget_check_and_reserve(ip, est, admin=False):
    """Reserve est tokens against the per-IP and global hourly
 budgets.
    Returns None if allowed, else an error string. Admins skip
 the per-IP cap
    but still trip the global circuit breaker (a runaway test
stops too)."""
    t = time.monotonic()
    with _rate_lock:
        if _window_sum(_global_usage, t) + est > TOKENS_GLOBAL
_HOUR:
            return "service paused — hourly usage budget reach
ed, please try later"
        if not admin:
            ipu = _ip_usage.setdefault(ip, [])
            if _window_sum(ipu, t) + est > TOKENS_PER_IP_HOUR:
                return "hourly usage limit reached — please tr
y again later"
            ipu.append((t, est))
        _global_usage.append((t, est))
    return None
